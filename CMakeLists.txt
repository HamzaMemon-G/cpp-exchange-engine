cmake_minimum_required(VERSION 3.20)
project(cpp_exchange_engine VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EXCHANGE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(exchange_core src/engine.cpp src/commands.cpp src/event_log.cpp src/workload.cpp)
target_include_directories(exchange_core PUBLIC include)
if(MSVC)
  target_compile_options(exchange_core PUBLIC /W4 /permissive-)
else()
  target_compile_options(exchange_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(EXCHANGE_SANITIZERS)
    target_compile_options(exchange_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(exchange_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(exchange_cli src/main.cpp)
target_link_libraries(exchange_cli PRIVATE exchange_core)
add_executable(exchange_bench src/benchmark.cpp)
target_link_libraries(exchange_bench PRIVATE exchange_core)
if(WIN32)
  target_link_libraries(exchange_bench PRIVATE psapi)
endif()
target_compile_definitions(exchange_bench PRIVATE EXCHANGE_BUILD_MODE="$<CONFIG>")
add_executable(exchange_generate src/generate.cpp)
target_link_libraries(exchange_generate PRIVATE exchange_core)
enable_testing()
add_executable(exchange_tests tests/tests.cpp)
target_link_libraries(exchange_tests PRIVATE exchange_core)
add_test(NAME exchange_tests COMMAND exchange_tests)
add_test(NAME cli_example COMMAND exchange_cli --input ${CMAKE_CURRENT_SOURCE_DIR}/examples/commands.txt)
add_test(NAME cli_integration COMMAND ${CMAKE_COMMAND} -DCLI=$<TARGET_FILE:exchange_cli> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
add_test(NAME benchmark_profiles COMMAND ${CMAKE_COMMAND} -DBENCH=$<TARGET_FILE:exchange_bench>
  -DGENERATOR=$<TARGET_FILE:exchange_generate> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark.cmake)
